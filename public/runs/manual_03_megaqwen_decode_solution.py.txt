"""MegaQwen decode — bf16 matmuls with float32 accumulation on AMD MI325X.

4x faster than float32 matmuls on gfx942 MFMA units.
RMSNorm, RoPE, and softmax stay in float32 for numerical stability.
Linear projections and attention use bf16 inputs with float32 accumulation.
"""
from __future__ import annotations
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

OP_TYPE = "megaqwen_decode"
SUPPORTED_PRECISIONS = ["bf16"]
HARDWARE_REQUIRED = ["MI325X"]

HIDDEN = 1024
INTERMEDIATE = 3072
NUM_Q = 16
NUM_KV = 8
HEAD_DIM = 128
NUM_LAYERS = 4
EPS = 1e-6


def _rmsnorm(x, w):
    var = x.pow(2).mean(dim=-1, keepdim=True)
    return x * torch.rsqrt(var + EPS) * w


def _rope(q, k, pos):
    D = q.shape[-1]
    half = D // 2
    inv = 1.0 / (10000 ** (torch.arange(0, half, device=q.device, dtype=torch.float32) / half))
    t = torch.tensor([pos], device=q.device, dtype=torch.float32)
    freqs = torch.outer(t, inv)
    cos, sin = freqs.cos(), freqs.sin()
    def apply(x):
        x1, x2 = x[..., :half], x[..., half:]
        return torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)
    return apply(q), apply(k)


class Block(nn.Module):
    def __init__(self):
        super().__init__()
        H, I, D = HIDDEN, INTERMEDIATE, HEAD_DIM
        self.input_ln = nn.Parameter(torch.ones(H, dtype=torch.bfloat16))
        self.q_proj = nn.Parameter(torch.empty(NUM_Q * D, H, dtype=torch.bfloat16))
        self.k_proj = nn.Parameter(torch.empty(NUM_KV * D, H, dtype=torch.bfloat16))
        self.v_proj = nn.Parameter(torch.empty(NUM_KV * D, H, dtype=torch.bfloat16))
        self.q_norm = nn.Parameter(torch.ones(D, dtype=torch.bfloat16))
        self.k_norm = nn.Parameter(torch.ones(D, dtype=torch.bfloat16))
        self.o_proj = nn.Parameter(torch.empty(H, NUM_Q * D, dtype=torch.bfloat16))
        self.post_ln = nn.Parameter(torch.ones(H, dtype=torch.bfloat16))
        self.gate_proj = nn.Parameter(torch.empty(I, H, dtype=torch.bfloat16))
        self.up_proj = nn.Parameter(torch.empty(I, H, dtype=torch.bfloat16))
        self.down_proj = nn.Parameter(torch.empty(H, I, dtype=torch.bfloat16))
        for p in self.parameters():
            if p is self.input_ln or p is self.post_ln or p is self.q_norm or p is self.k_norm:
                continue
            p.data.zero_()

    def forward(self, x, k_cache, v_cache, position):
        residual = x.float()
        h = _rmsnorm(residual, self.input_ln.float())
        h_bf16 = h.to(torch.bfloat16)

        # Linear projections in bf16 (4x faster than float32 on gfx942)
        q = (h_bf16 @ self.q_proj.T).float().view(NUM_Q, HEAD_DIM)
        k = (h_bf16 @ self.k_proj.T).float().view(NUM_KV, HEAD_DIM)
        v = (h_bf16 @ self.v_proj.T).float().view(NUM_KV, HEAD_DIM)

        q = _rmsnorm(q, self.q_norm.float())
        k = _rmsnorm(k, self.k_norm.float())
        q, k = _rope(q, k, position)

        k_cache = k_cache.clone()
        v_cache = v_cache.clone()
        k_cache[:, position, :] = k.to(k_cache.dtype)
        v_cache[:, position, :] = v.to(v_cache.dtype)

        # GQA attention: bf16 matmul, float32 softmax
        k_all = k_cache[:, :position + 1, :]  # (NUM_KV, L, D) bf16
        v_all = v_cache[:, :position + 1, :]  # (NUM_KV, L, D) bf16
        rep = NUM_Q // NUM_KV
        q_grouped = q.view(NUM_KV, rep, HEAD_DIM).to(torch.bfloat16)  # (NUM_KV, rep, D)
        scale = 1.0 / math.sqrt(HEAD_DIM)
        scores = torch.bmm(q_grouped, k_all.transpose(1, 2)).float() * scale  # (NUM_KV, rep, L)
        att = torch.softmax(scores, dim=-1)
        attn_out = torch.bmm(att.to(torch.bfloat16), v_all).float().reshape(-1)  # (NUM_Q * D,)

        # Output projection in bf16
        attn_out = (attn_out.to(torch.bfloat16) @ self.o_proj.T).float()

        h = residual + attn_out
        residual = h
        h = _rmsnorm(h, self.post_ln.float())
        h_bf16 = h.to(torch.bfloat16)

        # FFN in bf16
        gate = (h_bf16 @ self.gate_proj.T).float()
        up = (h_bf16 @ self.up_proj.T).float()
        h2 = F.silu(gate) * up
        h2 = (h2.to(torch.bfloat16) @ self.down_proj.T).float()

        y = residual + h2
        return y.to(torch.bfloat16), k_cache, v_cache


class Model(nn.Module):
    def __init__(self, num_layers=NUM_LAYERS, max_seq=131072):
        super().__init__()
        self.num_layers = num_layers
        self.max_seq = max_seq
        self.blocks = nn.ModuleList([Block() for _ in range(num_layers)])

    def forward(self, x, k_caches, v_caches, position):
        h = x
        new_k, new_v = [], []
        for i, block in enumerate(self.blocks):
            h, kc, vc = block(h, k_caches[i], v_caches[i], position)
            new_k.append(kc)
            new_v.append(vc)
        return h, new_k, new_v


def empty_caches(num_layers, max_seq, device, dtype=torch.bfloat16):
    k = [torch.zeros(NUM_KV, max_seq, HEAD_DIM, device=device, dtype=dtype) for _ in range(num_layers)]
    v = [torch.zeros(NUM_KV, max_seq, HEAD_DIM, device=device, dtype=dtype) for _ in range(num_layers)]
    return k, v


def _seeded_hidden(seed, device):
    g = torch.Generator(device="cpu")
    g.manual_seed(seed)
    return torch.randn(HIDDEN, generator=g, dtype=torch.bfloat16).to(device)


@torch.no_grad()
def prefill(model, ctx_len, seed, device=None):
    device = device or next(model.parameters()).device
    model = model.to(device).eval()
    assert ctx_len <= model.max_seq
    h = _seeded_hidden(seed, device)
    k_caches, v_caches = empty_caches(model.num_layers, model.max_seq, device)
    g = torch.Generator(device="cpu")
    g.manual_seed(seed + 1)
    for t in range(ctx_len):
        x_t = torch.randn(HIDDEN, generator=g, dtype=torch.bfloat16).to(device)
        x_t = (0.5 * x_t + 0.5 * h).to(torch.bfloat16)
        h, k_caches, v_caches = model(x_t, k_caches, v_caches, t)
    return h, k_caches, v_caches


@torch.no_grad()
def decode_steps(model, hidden, k_caches, v_caches, start_pos, n_steps, seed):
    device = hidden.device
    g = torch.Generator(device="cpu")
    g.manual_seed(seed + 2)
    h = hidden
    for i in range(n_steps):
        pos = start_pos + i
        x_t = torch.randn(HIDDEN, generator=g, dtype=torch.bfloat16).to(device)
        x_t = (0.5 * x_t + 0.5 * h).to(torch.bfloat16)
        h, k_caches, v_caches = model(x_t, k_caches, v_caches, pos)
    return h, k_caches, v_caches


def run(ctx_len, n_decode, seed, model=None, max_seq=None):
    device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
    max_seq = max_seq or max(ctx_len + n_decode, 512)
    if model is None:
        model = Model(NUM_LAYERS, max_seq)
    else:
        if getattr(model, "max_seq", 0) < ctx_len + n_decode:
            raise ValueError(f"model.max_seq={getattr(model, 'max_seq', None)} too small")
    model = model.to(device).eval()
    h, k_caches, v_caches = prefill(model, ctx_len, seed, device)
    h, k_caches, v_caches = decode_steps(model, h, k_caches, v_caches, ctx_len, n_decode, seed)
    return h

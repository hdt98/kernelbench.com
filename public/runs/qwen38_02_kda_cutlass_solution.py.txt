"""Kimi Delta Attention chunked forward for MI325X -- Triton + batched matmul.

Qwen 3.8 (Nexus-hosted) in-session solution for KernelBench-AMD 02_kda_cutlass.

Reproduces the reference chunk-form KDA forward exactly (fp32 math, bf16 in
and out), with the two O(BT)-long python loops of the naive version replaced
by closed forms:

  * The intra-chunk K.K interaction
        A[c, i] = <k_c odot exp(g_c - g_i), k_i>
    is one batched matmul: (k odot exp(g)) @ (k odot exp(-g))^T, computed by
    a Triton kernel (fp32 ieee throughout, one program per (b, h, chunk)
    tile). The same kernel builds the Q.K attention tile of the inter-chunk
    pass; _pair_tiles enforces contiguous inputs because the kernel uses
    flat pid*C*K batch offsets.

  * The delta-rule triangular solve loop
        A[i, :i] += sum_m A[i, m] A[m, :i]   (row-ascending)
    computes F = (I - L)^-1 L for the strict-lower L, so A + I = (I - L)^-1.
    Because L^BT = 0, the inverse is the finite Neumann sum
        I + L + L^2 + ... + L^(BT-1),
    evaluated by 5 squaring doublings of 64x64 tiles (exact in fp32 for these
    magnitudes; verified against the reference loop).

The inter-chunk recurrence stays sequential over the <= T/64 chunks but is
fully vectorized over (B, H): each step is a handful of batched matmuls on
(B*H, ...) tensors. CUDA-graph-cached on the input pointers.
"""
from __future__ import annotations

import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def _scaled_pair_dot(a_ptr, b_ptr, g_ptr, out_ptr,
                     K: tl.constexpr, C: tl.constexpr):
    # out[c, i] = sum_d a[c, d] * exp(+g[c, d]) * b[i, d] * exp(-g[i, d])
    pid = tl.program_id(0)
    rc = tl.arange(0, C)
    rd = tl.arange(0, K)
    base = pid * C * K
    a = tl.load(a_ptr + base + rc[:, None] * K + rd[None, :]).to(tl.float32)
    b = tl.load(b_ptr + base + rc[:, None] * K + rd[None, :]).to(tl.float32)
    g = tl.load(g_ptr + base + rc[:, None] * K + rd[None, :])
    ea = a * tl.exp(g)
    eb = b * tl.exp(-g)
    out = tl.dot(ea, tl.trans(eb), input_precision="ieee")
    tl.store(out_ptr + pid * C * C + rc[:, None] * C + rc[None, :], out)


def _rearrange(x: torch.Tensor, chunk: int) -> torch.Tensor:
    B, T, H, D = x.shape
    return x.permute(0, 2, 1, 3).reshape(B, H, T // chunk, chunk, D).contiguous()


def _pair_tiles(a: torch.Tensor, b: torch.Tensor, g: torch.Tensor,
                C: int) -> torch.Tensor:
    """(B,H,N,C,K) tensors -> (B,H,N,C,C) fp32 pair tiles via the kernel."""
    # The kernel addresses batches at flat pid*C*K offsets: inputs MUST be
    # contiguous (per-chunk slices like q[:, :, i] are strided views).
    a, b, g = a.contiguous(), b.contiguous(), g.contiguous()
    B, H, N, _, K = a.shape
    out = torch.empty(B, H, N, C, C, dtype=torch.float32, device=a.device)
    _scaled_pair_dot[(B * H * N,)](a, b, g, out, K=K, C=C, num_warps=4)
    return out


def _neumann_inverse_minus_i(L: torch.Tensor, C: int) -> torch.Tensor:
    """(I - L)^-1 - I for strict-lower L with L^C = 0: sum_{i=1}^{C-1} L^i."""
    I = torch.eye(C, dtype=L.dtype, device=L.device)
    P = I + L                    # covers L^0 .. L^1
    Q = L @ L                    # next power: L^2
    steps = (C.bit_length() - 1) - 1   # doublings to reach L^(C-1)
    for _ in range(steps):
        P = P + P @ Q
        Q = Q @ Q
    return P - I


class Model(nn.Module):
    """KDA forward (chunk form). No learned parameters; activations only."""

    def __init__(self, B: int, T: int, H: int, K: int, V: int, chunk_size: int = 64):  # noqa: E741
        super().__init__()
        self.B, self.T, self.H = B, T, H
        self.K, self.V = K, V
        self.chunk_size = chunk_size
        self.scale = float(K) ** -0.5
        self.register_buffer("_dummy", torch.zeros(1), persistent=False)
        self._graph = None
        self._graph_key = None
        self._graph_out = None

    def _compute(self, q, k, v, g, beta):
        chunk = self.chunk_size
        B, T, H, K = q.shape
        V = v.shape[-1]
        N = T // chunk

        q = _rearrange(q, chunk).float() * self.scale
        k = _rearrange(k, chunk).float()
        v = _rearrange(v, chunk).float()
        g = _rearrange(g, chunk).float()
        g = g.cumsum(-2)
        beta = beta.permute(0, 2, 1).reshape(B, H, N, chunk).float()

        # --- intra-chunk delta-rule factorization ---
        A = _pair_tiles(k, k, g, chunk)                  # (B,H,N,C,C) fp32
        A = A * beta[..., None]                          # row scale by beta_c
        mask_upper = torch.triu(torch.ones(chunk, chunk, dtype=torch.bool,
                                           device=q.device), diagonal=0)
        L = -A.masked_fill(mask_upper, 0)                # strict lower
        Ainv = _neumann_inverse_minus_i(L, chunk) + torch.eye(
            chunk, dtype=torch.float32, device=q.device)
        A = Ainv * beta[..., None, :]                    # column scale

        kg = (g.exp() * k)
        w = A @ kg                                        # (B,H,N,C,K)
        u = A @ v                                         # (B,H,N,C,V)

        # --- inter-chunk recurrence (sequential over chunks, batched BH) ---
        S = q.new_zeros(B, H, K, V)
        o = torch.empty_like(v)
        mask_su = torch.triu(torch.ones(chunk, chunk, dtype=torch.bool,
                                        device=q.device), diagonal=1)
        qeg = q * g.exp()
        for i in range(N):
            q_i = q[:, :, i]
            k_i = k[:, :, i]
            g_i = g[:, :, i]
            Aqk = _pair_tiles(q_i.unsqueeze(2), k_i.unsqueeze(2),
                              g_i.unsqueeze(2), chunk).squeeze(2)
            Aqk = Aqk.masked_fill(mask_su, 0)             # keep c >= j
            v_i = u[:, :, i] - w[:, :, i] @ S
            o[:, :, i] = qeg[:, :, i] @ S + Aqk @ v_i
            gl = g_i[:, :, -1]                            # (B,H,K)
            S = S * gl.exp()[..., None]
            S = S + (torch.exp(gl[:, :, None, :] - g_i).transpose(-1, -2)
                     * k_i.transpose(-1, -2)) @ v_i

        o = o.reshape(B, H, T, V).permute(0, 2, 1, 3)
        return o.to(torch.bfloat16)

    def forward(self, q, k, v, g, beta):
        key = (q.data_ptr(), k.data_ptr(), v.data_ptr(), g.data_ptr(),
               beta.data_ptr(), tuple(q.shape))
        if self._graph is not None and self._graph_key == key:
            self._graph.replay()
            return self._graph_out
        s = torch.cuda.Stream()
        s.wait_stream(torch.cuda.current_stream())
        with torch.cuda.stream(s):
            for _ in range(2):
                out = self._compute(q, k, v, g, beta)
        torch.cuda.current_stream().wait_stream(s)
        gr = torch.cuda.CUDAGraph()
        with torch.cuda.graph(gr):
            out = self._compute(q, k, v, g, beta)
        self._graph = gr
        self._graph_key = key
        self._graph_out = out
        gr.replay()
        return out


# Same default shape and input distribution as the reference deck entry
# (standalone shim; the reference module is not imported).
B = 2
T = 1024
H = 8
K = 128
V = 128
CHUNK_SIZE = 64


def get_inputs():
    torch.manual_seed(0)
    q = torch.randn(B, T, H, K, dtype=torch.bfloat16) * 0.1
    k = torch.randn(B, T, H, K, dtype=torch.bfloat16) * 0.1
    v = torch.randn(B, T, H, V, dtype=torch.bfloat16) * 0.1
    g = (torch.randn(B, T, H, K, dtype=torch.float32) * 0.1 - 0.05)
    beta = torch.sigmoid(torch.randn(B, T, H, dtype=torch.bfloat16))
    return [q, k, v, g, beta]


def get_init_inputs():
    return [B, T, H, K, V, CHUNK_SIZE]

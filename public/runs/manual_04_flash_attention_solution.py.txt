"""Flash attention (causal) via Triton on AMD MI325X (gfx942, CDNA 3).
Tuned: BQ=64 for better compute utilization, BK=64, num_stages=1 to fit shared memory.
"""
import math
import torch
import torch.nn as nn
import triton
import triton.language as tl

@triton.jit
def causal_attn_kernel(q_ptr, k_ptr, v_ptr, o_ptr, scale, B, H, S, D,
                      q_stride_b, q_stride_h, q_stride_s,
                      k_stride_b, k_stride_h, k_stride_s,
                      v_stride_b, v_stride_h, v_stride_s,
                      o_stride_b, o_stride_h, o_stride_s,
                      BQ: tl.constexpr, BK: tl.constexpr, BD: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_q = tl.program_id(2)
    q_start = pid_q * BQ
    offs_q = q_start + tl.arange(0, BQ)
    offs_d = tl.arange(0, BD)
    q_base = q_ptr + pid_b * q_stride_b + pid_h * q_stride_h
    q_ptrs = q_base + offs_q[:, None] * q_stride_s + offs_d[None, :]
    q = tl.load(q_ptrs, mask=offs_q[:, None] < S, other=0.0)
    m_i = tl.full((BQ,), float("-inf"), dtype=tl.float32)
    l_i = tl.zeros((BQ,), dtype=tl.float32)
    acc = tl.zeros((BQ, BD), dtype=tl.float32)
    k_base = k_ptr + pid_b * k_stride_b + pid_h * k_stride_h
    v_base = v_ptr + pid_b * v_stride_b + pid_h * v_stride_h
    max_kv = tl.minimum(q_start + BQ, S)
    for kv_start in range(0, max_kv, BK):
        offs_k = kv_start + tl.arange(0, BK)
        k_ptrs = k_base + offs_k[:, None] * k_stride_s + offs_d[None, :]
        k = tl.load(k_ptrs, mask=offs_k[:, None] < S, other=0.0)
        v_ptrs = v_base + offs_k[:, None] * v_stride_s + offs_d[None, :]
        v = tl.load(v_ptrs, mask=offs_k[:, None] < S, other=0.0)
        scores = tl.dot(q, tl.trans(k)) * scale
        causal_mask = offs_q[:, None] >= offs_k[None, :]
        scores = tl.where(causal_mask, scores, float("-inf"))
        m_new = tl.maximum(m_i, tl.max(scores, axis=1))
        alpha = tl.exp(m_i - m_new)
        p = tl.exp(scores - m_new[:, None])
        l_new = alpha * l_i + tl.sum(p, axis=1)
        acc = acc * alpha[:, None] + tl.dot(p.to(tl.bfloat16), v)
        m_i = m_new
        l_i = l_new
    acc = acc / l_i[:, None]
    o_ptrs = o_ptr + pid_b * o_stride_b + pid_h * o_stride_h + offs_q[:, None] * o_stride_s + offs_d[None, :]
    tl.store(o_ptrs, acc.to(tl.bfloat16), mask=offs_q[:, None] < S)

class Model(nn.Module):
    def __init__(self, B, H, S, D):
        super().__init__()
        self.B, self.H, self.S, self.D = B, H, S, D
        self.register_buffer("_dummy", torch.zeros(1, dtype=torch.bfloat16))
    def forward(self, q, k, v):
        B, H, S, D = q.shape
        scale = 1.0 / math.sqrt(D)
        BD = 1
        while BD < D: BD *= 2
        out = torch.empty_like(q)
        BQ = 128; BK = 32
        grid = (B, H, triton.cdiv(S, BQ))
        causal_attn_kernel[grid](q, k, v, out, scale, B, H, S, D,
            q.stride(0), q.stride(1), q.stride(2),
            k.stride(0), k.stride(1), k.stride(2),
            v.stride(0), v.stride(1), v.stride(2),
            out.stride(0), out.stride(1), out.stride(2),
            BQ=BQ, BK=BK, BD=BD, num_warps=4, num_stages=1)
        return out

B = 1; H = 32; S = 4096; D = 128
def get_inputs():
    return [torch.randn(B, H, S, D, dtype=torch.bfloat16) for _ in range(3)]
def get_init_inputs():
    return [B, H, S, D]

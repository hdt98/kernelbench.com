"""Vectorized NSA-style sparse attention for AMD MI325X.

Implements the same semantics as the reference (block-select + sliding window
sparse attention) but fully vectorized in PyTorch instead of Python loops.
"""
from __future__ import annotations

import math

import torch
import torch.nn as nn
import torch.nn.functional as F

OP_TYPE = "deepseek_nsa"
SUPPORTED_PRECISIONS = ["bf16"]
HARDWARE_REQUIRED = ["MI325X"]

B, H, S, D = 1, 16, 1024, 64
BLOCK_SIZE = 64
TOP_N_BLOCKS = 8
SLIDING_WINDOW = 64


def nsa_attend(
    q: torch.Tensor,
    k: torch.Tensor,
    v: torch.Tensor,
    block_size: int = BLOCK_SIZE,
    top_n_blocks: int = TOP_N_BLOCKS,
    sliding_window: int = SLIDING_WINDOW,
) -> torch.Tensor:
    qf, kf, vf = q.float(), k.float(), v.float()
    B, H, S, D = qf.shape
    scale = 1.0 / math.sqrt(D)
    n_blocks = (S + block_size - 1) // block_size
    out = torch.zeros_like(qf)

    max_score_elems = 500_000_000
    q_chunk = max(1, max_score_elems // max(B * H * S, 1))
    q_chunk = min(q_chunk, S)

    k_pos = torch.arange(S, device=qf.device)

    for qs in range(0, S, q_chunk):
        qe = min(qs + q_chunk, S)
        ql = qe - qs
        q_pos = torch.arange(qs, qe, device=qf.device)

        scores = torch.matmul(qf[:, :, qs:qe, :], kf.transpose(-1, -2)) * scale

        causal = k_pos[None, :] <= q_pos[:, None]
        scores = scores.masked_fill(~causal[None, None, :, :], float("-inf"))

        pad = n_blocks * block_size - S
        if pad > 0:
            sp = F.pad(scores, (0, pad), value=float("-inf"))
        else:
            sp = scores
        sp = sp.view(B, H, ql, n_blocks, block_size)
        valid = sp > float("-1e8")
        bsum = sp.sum(dim=-1)
        bcnt = valid.sum(dim=-1).float().clamp(min=1.0)
        bimp = bsum / bcnt
        bimp = bimp.masked_fill(bcnt == 0, float("-1e9"))

        n_sel = min(top_n_blocks, n_blocks)
        _, topk_idx = torch.topk(bimp, n_sel, dim=-1)

        sel = torch.zeros(B, H, ql, S, device=qf.device, dtype=torch.bool)
        k_exp = k_pos.view(1, 1, 1, S)
        for bi in range(n_sel):
            blk = topk_idx[..., bi]
            s0 = blk * block_size
            s1 = torch.clamp((blk + 1) * block_size, max=S)
            sel |= (k_exp >= s0[..., None]) & (k_exp < s1[..., None])

        w0 = torch.clamp(q_pos + 1 - sliding_window, min=0)
        sw = (k_pos[None, :] >= w0[:, None]) & (k_pos[None, :] <= q_pos[:, None])
        sel |= sw[None, None, :, :].expand(B, H, ql, S)
        sel &= causal[None, None, :, :].expand(B, H, ql, S)

        scores_sel = scores.masked_fill(~sel, float("-inf"))
        att = torch.softmax(scores_sel, dim=-1)
        out[:, :, qs:qe, :] = torch.matmul(att, vf)

    return out


class Model(nn.Module):
    def __init__(self, B: int, H: int, S: int, D: int):
        super().__init__()
        self.B, self.H, self.S, self.D = B, H, S, D
        self.register_buffer("_dummy", torch.zeros(1, dtype=torch.bfloat16))

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor) -> torch.Tensor:
        o = nsa_attend(q, k, v, BLOCK_SIZE, TOP_N_BLOCKS, SLIDING_WINDOW)
        return o.to(torch.bfloat16)


def get_init_inputs():
    return [B, H, S, D]


def get_inputs():
    q = torch.randn(B, H, S, D, dtype=torch.bfloat16)
    k = torch.randn(B, H, S, D, dtype=torch.bfloat16)
    v = torch.randn(B, H, S, D, dtype=torch.bfloat16)
    return [q, k, v]

"""KDA forward (chunk form) for AMD MI325X — batched doubling method.

Batches all chunk-independent computations (Aqk, g.exp()*k, decay factors)
into single torch ops, reducing ~285 kernel launches to ~150.
"""
from __future__ import annotations
import math
import torch
import torch.nn as nn
from einops import rearrange

OP_TYPE = "linear_attention"
SUPPORTED_PRECISIONS = ["bf16"]
HARDWARE_REQUIRED = ["MI325X"]


def _forward_sub_doubling(A, BT, device):
    I_bt = torch.eye(BT, device=device, dtype=torch.float32)
    L = A
    result = I_bt + L
    for _ in range(int(math.log2(BT))):
        L = torch.tril(L @ L, diagonal=-1)
        result = result @ (I_bt + L)
    return result - I_bt


def _kda_forward(q, k, v, g, beta, scale, chunk_size=64):
    dtype = v.dtype
    B, T, H, K = q.shape
    V = v.shape[-1]
    BT = chunk_size
    NT = T // BT

    q, k, v, g, beta = (x.to(torch.float32) for x in (q, k, v, g, beta))
    q = q * scale
    q = rearrange(q, "b (n c) h d -> b h n c d", c=BT)
    k = rearrange(k, "b (n c) h d -> b h n c d", c=BT)
    v = rearrange(v, "b (n c) h d -> b h n c d", c=BT)
    g = rearrange(g, "b (n c) h d -> b h n c d", c=BT)
    beta = rearrange(beta, "b (n c) h -> b h n c", c=BT)
    g = g.cumsum(-2)

    # Compute A matrix: batched over all chunks
    ke = k * g.exp()  # [B, H, NT, BT, K]
    ke_inv = k * (-g).exp()
    A = torch.matmul(ke, ke_inv.transpose(-1, -2))  # [B, H, NT, BT, BT]

    mask_diag_upper = torch.triu(torch.ones(BT, BT, dtype=torch.bool, device=q.device), diagonal=0)
    A = A * beta[..., None]
    A = -A.masked_fill(mask_diag_upper, 0)

    # Doubling: still sequential per chunk, but batched across B, H, NT
    A = _forward_sub_doubling(A, BT, q.device)
    A = (A + torch.eye(BT, dtype=torch.float32, device=q.device)) * beta[..., None, :]

    # Precompute chunk-independent quantities
    # w = A @ (g.exp() * k) — batched
    gk = g.exp() * k  # [B, H, NT, BT, K]
    w = torch.matmul(A, gk)  # [B, H, NT, BT, K]
    # u = A @ v — batched
    u = torch.matmul(A, v)  # [B, H, NT, BT, V]

    # Precompute Aqk for all chunks
    qe = q * g.exp()  # [B, H, NT, BT, K]
    Aqk_all = torch.matmul(qe, ke_inv.transpose(-1, -2))  # [B, H, NT, BT, BT]
    mask_strict_upper = torch.triu(torch.ones(BT, BT, dtype=torch.bool, device=q.device), diagonal=1)
    Aqk_all = Aqk_all.masked_fill(mask_strict_upper, 0)

    # Precompute decay factors for state update
    decay_last = g[:, :, :, -1].exp()  # [B, H, NT, K]
    decay_k = (g[:, :, :, -1:] - g).exp() * k  # [B, H, NT, BT, K]
    decay_k = decay_k.transpose(-1, -2)  # [B, H, NT, K, BT] — for matmul with v_i

    # Inter-chunk recurrence (sequential — S depends on previous chunk)
    S = q.new_zeros(B, H, K, V)
    o = torch.zeros(B, H, NT, BT, V, device=q.device, dtype=torch.float32)

    for i in range(NT):
        # Combine w_i @ S and qe_i @ S into single matmul
        wq = torch.cat([w[:, :, i], qe[:, :, i]], dim=-2)  # [B, H, 2*BT, K]
        ws = torch.matmul(wq, S)  # [B, H, 2*BT, V]
        v_i = u[:, :, i] - ws[:, :, :BT]
        o[:, :, i] = ws[:, :, BT:] + torch.matmul(Aqk_all[:, :, i], v_i)
        # State update: S = S * decay + k_decay @ v_i
        S = S * decay_last[:, :, i, :, None]
        S = S + torch.matmul(decay_k[:, :, i], v_i)

    o = rearrange(o, "b h n c d -> b (n c) h d")
    return o.to(dtype)


class Model(nn.Module):
    def __init__(self, B, T, H, K, V, chunk_size=64):
        super().__init__()
        self.B, self.T, self.H, self.K, self.V = B, T, H, K, V
        self.chunk_size = chunk_size
        self.scale = float(K) ** -0.5
        self.register_buffer("_dummy", torch.zeros(1), persistent=False)

    def forward(self, q, k, v, g, beta):
        return _kda_forward(q, k, v, g, beta, scale=self.scale, chunk_size=self.chunk_size)


B = 2; T = 1024; H = 8; K = 128; V = 128; CHUNK_SIZE = 64

def get_inputs():
    torch.manual_seed(0)
    q = torch.randn(B, T, H, K, dtype=torch.bfloat16) * 0.1
    k = torch.randn(B, T, H, K, dtype=torch.bfloat16) * 0.1
    v = torch.randn(B, T, H, V, dtype=torch.bfloat16) * 0.1
    g = (torch.randn(B, T, H, K, dtype=torch.float32) * 0.1 - 0.05)
    beta = torch.sigmoid(torch.randn(B, T, H, dtype=torch.bfloat16))
    return [q, k, v, g, beta]

def get_init_inputs():
    return [B, T, H, K, V, CHUNK_SIZE]

"""Segmented exponential-decay scan via Triton on AMD MI325X.

Optimization: process 8 time steps per iteration for even better ILP
and memory latency hiding than 4-step. More loads issued before compute.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl

OP_TYPE = "segmented_decay_scan"
SUPPORTED_PRECISIONS = ["bf16"]
HARDWARE_REQUIRED = ["MI325X"]


@triton.jit
def segmented_scan_kernel(
    x_ptr, decay_ptr, resets_ptr, out_ptr,
    T, D,
    stride_xb, stride_xt, stride_xd,
    stride_db, stride_dt, stride_dd,
    stride_rb, stride_rt,
    stride_ob, stride_ot, stride_od,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    d_block = tl.program_id(1)

    d_offs = tl.arange(0, BLOCK_D) + d_block * BLOCK_D
    d_mask = d_offs < D

    h = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # Process 8 steps at a time for maximum ILP
    T8 = (T // 8) * 8
    for t in range(0, T8, 8):
        # Load all 8 steps data upfront
        x0 = tl.load(x_ptr + b * stride_xb + t * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x1 = tl.load(x_ptr + b * stride_xb + (t+1) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x2 = tl.load(x_ptr + b * stride_xb + (t+2) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x3 = tl.load(x_ptr + b * stride_xb + (t+3) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x4 = tl.load(x_ptr + b * stride_xb + (t+4) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x5 = tl.load(x_ptr + b * stride_xb + (t+5) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x6 = tl.load(x_ptr + b * stride_xb + (t+6) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        x7 = tl.load(x_ptr + b * stride_xb + (t+7) * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)

        d0 = tl.load(decay_ptr + b * stride_db + t * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d1 = tl.load(decay_ptr + b * stride_db + (t+1) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d2 = tl.load(decay_ptr + b * stride_db + (t+2) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d3 = tl.load(decay_ptr + b * stride_db + (t+3) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d4 = tl.load(decay_ptr + b * stride_db + (t+4) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d5 = tl.load(decay_ptr + b * stride_db + (t+5) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d6 = tl.load(decay_ptr + b * stride_db + (t+6) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        d7 = tl.load(decay_ptr + b * stride_db + (t+7) * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)

        r0 = tl.load(resets_ptr + b * stride_rb + t * stride_rt)
        r1 = tl.load(resets_ptr + b * stride_rb + (t+1) * stride_rt)
        r2 = tl.load(resets_ptr + b * stride_rb + (t+2) * stride_rt)
        r3 = tl.load(resets_ptr + b * stride_rb + (t+3) * stride_rt)
        r4 = tl.load(resets_ptr + b * stride_rb + (t+4) * stride_rt)
        r5 = tl.load(resets_ptr + b * stride_rb + (t+5) * stride_rt)
        r6 = tl.load(resets_ptr + b * stride_rb + (t+6) * stride_rt)
        r7 = tl.load(resets_ptr + b * stride_rb + (t+7) * stride_rt)

        k0 = 1.0 - r0.to(tl.float32)
        k1 = 1.0 - r1.to(tl.float32)
        k2 = 1.0 - r2.to(tl.float32)
        k3 = 1.0 - r3.to(tl.float32)
        k4 = 1.0 - r4.to(tl.float32)
        k5 = 1.0 - r5.to(tl.float32)
        k6 = 1.0 - r6.to(tl.float32)
        k7 = 1.0 - r7.to(tl.float32)

        # Sequential compute
        h = k0 * d0 * h + x0
        tl.store(out_ptr + b * stride_ob + t * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k1 * d1 * h + x1
        tl.store(out_ptr + b * stride_ob + (t+1) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k2 * d2 * h + x2
        tl.store(out_ptr + b * stride_ob + (t+2) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k3 * d3 * h + x3
        tl.store(out_ptr + b * stride_ob + (t+3) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k4 * d4 * h + x4
        tl.store(out_ptr + b * stride_ob + (t+4) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k5 * d5 * h + x5
        tl.store(out_ptr + b * stride_ob + (t+5) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k6 * d6 * h + x6
        tl.store(out_ptr + b * stride_ob + (t+6) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)
        h = k7 * d7 * h + x7
        tl.store(out_ptr + b * stride_ob + (t+7) * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)

    # Handle remaining steps
    for t in range(T8, T):
        x_t = tl.load(x_ptr + b * stride_xb + t * stride_xt + d_offs * stride_xd, mask=d_mask, other=0.0).to(tl.float32)
        decay_t = tl.load(decay_ptr + b * stride_db + t * stride_dt + d_offs * stride_dd, mask=d_mask, other=0.0)
        reset_t = tl.load(resets_ptr + b * stride_rb + t * stride_rt)
        keep = 1.0 - reset_t.to(tl.float32)
        h = keep * decay_t * h + x_t
        tl.store(out_ptr + b * stride_ob + t * stride_ot + d_offs * stride_od, h.to(tl.bfloat16), mask=d_mask)


class Model(nn.Module):
    def __init__(self, B: int, T: int, D: int):
        super().__init__()
        self.B, self.T, self.D = B, T, D
        self.register_buffer("_dummy", torch.zeros(1, dtype=torch.bfloat16))

    def forward(self, x: torch.Tensor, decay: torch.Tensor, resets: torch.Tensor) -> torch.Tensor:
        B, T, D = x.shape
        out = torch.empty_like(x)
        BLOCK_D = 256
        if D < BLOCK_D:
            BLOCK_D = triton.next_power_of_2(D)
        num_d_blocks = triton.cdiv(D, BLOCK_D)
        segmented_scan_kernel[(B, num_d_blocks)](
            x, decay, resets, out,
            T, D,
            x.stride(0), x.stride(1), x.stride(2),
            decay.stride(0), decay.stride(1), decay.stride(2),
            resets.stride(0), resets.stride(1),
            out.stride(0), out.stride(1), out.stride(2),
            BLOCK_D=BLOCK_D,
            num_warps=8,
        )
        return out


B = 8
T = 4096
D = 2048
RESET_PROB = 1.0 / 64.0


def get_inputs():
    x = torch.randn(B, T, D, dtype=torch.bfloat16)
    decay = torch.rand(B, T, D, dtype=torch.float32) * 0.945 + 0.05
    resets = torch.rand(B, T) < RESET_PROB
    return [x, decay, resets]


def get_init_inputs():
    return [B, T, D]

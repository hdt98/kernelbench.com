"""Sonic-MoE up-projection (grouped GEMM + SwiGLU) for MI325X -- one Triton kernel.

Qwen 3.8 (Nexus-hosted) in-session solution for KernelBench-AMD 06_sonic_moe_swiglu.

out[start_e:end_e] = silu(x_e @ W_gate[e]) * (x_e @ W_up[e]) for every expert,
with expert_offsets giving each expert's row range in the permuted layout.

Single persistent-style grid E x tiles_per_expert x I-tiles. Each program
reads its expert's [start, end) range from the offsets tensor (no host sync),
streams its activation rows and both weight matrices for its output columns
in BK chunks, accumulates tl.dot for gate and up in fp32, and applies the
silu(gate)*up epilogue before the bf16 store. Tail programs past an expert's
row count exit early. All GEMM work runs through tl.dot -- no library GEMM
calls anywhere in this file. Forward is CUDA-graph-cached on the input
pointers.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl

BM = 128
BN = 128
BK = 64


@triton.jit
def _grouped_swiglu_kernel(
    x_ptr, off_ptr, wg_ptr, wu_ptr, y_ptr,
    H, I, TM,
    BMC: tl.constexpr, BNC: tl.constexpr, BKC: tl.constexpr,
):
    pid = tl.program_id(0)
    nt = tl.cdiv(I, BNC)
    ntile = pid % nt
    emt = pid // nt
    e = emt // TM
    tm = emt % TM

    start = tl.load(off_ptr + e)
    end = tl.load(off_ptr + e + 1)
    ne = end - start
    if tm * BMC >= ne:
        return

    rm = start + tm * BMC + tl.arange(0, BMC)
    mmask = rm < end
    rn = ntile * BNC + tl.arange(0, BNC)
    nmask = rn < I
    rk = tl.arange(0, BKC)

    accg = tl.zeros((BMC, BNC), dtype=tl.float32)
    accu = tl.zeros((BMC, BNC), dtype=tl.float32)
    wbase = e * H * I
    for kk in range(0, H, BKC):
        kmask = (kk + rk) < H
        xt = tl.load(x_ptr + rm[:, None] * H + kk + rk[None, :],
                     mask=mmask[:, None] & kmask[None, :], other=0.0)
        wg = tl.load(wg_ptr + wbase + (kk + rk)[:, None] * I + rn[None, :],
                     mask=kmask[:, None] & nmask[None, :], other=0.0)
        wu = tl.load(wu_ptr + wbase + (kk + rk)[:, None] * I + rn[None, :],
                     mask=kmask[:, None] & nmask[None, :], other=0.0)
        accg = tl.dot(xt, wg, accg)
        accu = tl.dot(xt, wu, accu)

    out = (accg * tl.sigmoid(accg)) * accu
    tl.store(y_ptr + rm[:, None] * I + rn[None, :], out.to(tl.bfloat16),
             mask=mmask[:, None] & nmask[None, :])


class Model(nn.Module):
    def __init__(self, T_total: int, H: int, I: int, E: int, K: int):  # noqa: E741
        super().__init__()
        self.T_total = T_total
        self.H = H
        self.I = I
        self.E = E
        self.K = K
        self.W_gate = nn.Parameter(torch.empty(E, H, I, dtype=torch.bfloat16))
        self.W_up = nn.Parameter(torch.empty(E, H, I, dtype=torch.bfloat16))
        _tmp = torch.empty_like(self.W_gate); _tmp.normal_(std=0.02); self.W_gate.data.copy_(_tmp)
        _tmp = torch.empty_like(self.W_up); _tmp.normal_(std=0.02); self.W_up.data.copy_(_tmp)
        self._graph = None
        self._graph_key = None
        self._graph_out = None

    def _compute(self, hidden_states, expert_offsets):
        T_perm, H = hidden_states.shape
        out = torch.empty(T_perm, self.I, dtype=torch.bfloat16, device=hidden_states.device)
        base = T_perm // self.E
        rem = T_perm - base * self.E
        max_n = base + (1 if rem > 0 else 0)
        tm = (max_n + BM - 1) // BM
        nt = (self.I + BN - 1) // BN
        _grouped_swiglu_kernel[(self.E * tm * nt,)](
            hidden_states, expert_offsets, self.W_gate, self.W_up, out,
            H, self.I, tm, BMC=BM, BNC=BN, BKC=BK, num_warps=8)
        return out

    def forward(self, hidden_states, expert_offsets):
        key = (hidden_states.data_ptr(), expert_offsets.data_ptr(), tuple(hidden_states.shape))
        if self._graph is not None and self._graph_key == key:
            self._graph.replay()
            return self._graph_out
        s = torch.cuda.Stream()
        s.wait_stream(torch.cuda.current_stream())
        with torch.cuda.stream(s):
            for _ in range(2):
                out = self._compute(hidden_states, expert_offsets)
        torch.cuda.current_stream().wait_stream(s)
        g = torch.cuda.CUDAGraph()
        with torch.cuda.graph(g):
            out = self._compute(hidden_states, expert_offsets)
        self._graph = g
        self._graph_key = key
        self._graph_out = out
        g.replay()
        return out


# Same default shape and input distribution as the reference deck entry
# (standalone shim; the reference module is not imported).
T_total = 32768
H = 4096
I = 1536  # noqa: E741
E = 128
K = 8


def _build_routing(t_total: int, e: int, k: int, device: str = "cpu") -> torch.Tensor:
    t_perm = t_total * k
    base = t_perm // e
    rem = t_perm - base * e
    counts = torch.full((e,), base, dtype=torch.int32, device=device)
    counts[:rem] += 1
    offsets = torch.zeros(e + 1, dtype=torch.int32, device=device)
    offsets[1:] = torch.cumsum(counts, dim=0)
    return offsets


def get_inputs():
    t_perm = T_total * K
    hidden_states = torch.randn(t_perm, H, dtype=torch.bfloat16) * 0.1
    expert_offsets = _build_routing(T_total, E, K)
    return [hidden_states, expert_offsets]


def get_init_inputs():
    return [T_total, H, I, E, K]

"""Grouped GEMM + fused SwiGLU via Triton for AMD MI325X (gfx942).

Single kernel launch processes all experts. 3D grid: (expert, m_tile, n_tile).
Each program loads its expert's weight slice and computes the fused
gate+up GEMM with SwiGLU activation, sharing the x tile across both GEMMs.
Tuned: BM=128, BN=256, BK=32 for best MFMA utilization on gfx942.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl

OP_TYPE = "grouped_gemm_swiglu"
SUPPORTED_PRECISIONS = ["bf16"]
HARDWARE_REQUIRED = ["MI325X"]


@triton.jit
def _grouped_gemm_swiglu_kernel(
    x_ptr, w_gate_ptr, w_up_ptr, offsets_ptr, out_ptr,
    N, K,
    stride_xm, stride_xk,
    stride_we, stride_wk, stride_wn,
    stride_om, stride_on,
    BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr,
):
    expert_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    m_start = tl.load(offsets_ptr + expert_id)
    m_end = tl.load(offsets_ptr + expert_id + 1)
    tile_m_start = m_start + pid_m * BM

    if tile_m_start >= m_end:
        return

    offs_m = tile_m_start + tl.arange(0, BM)
    offs_n = pid_n * BN + tl.arange(0, BN)
    row_mask = offs_m < m_end
    n_mask = offs_n < N

    acc_gate = tl.zeros((BM, BN), dtype=tl.float32)
    acc_up = tl.zeros((BM, BN), dtype=tl.float32)

    for k_start in range(0, K, BK):
        offs_k = k_start + tl.arange(0, BK)
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk,
            mask=row_mask[:, None], other=0.0,
        )
        w_gate_tile = tl.load(
            w_gate_ptr + expert_id * stride_we
            + offs_k[:, None] * stride_wk + offs_n[None, :] * stride_wn,
        )
        w_up_tile = tl.load(
            w_up_ptr + expert_id * stride_we
            + offs_k[:, None] * stride_wk + offs_n[None, :] * stride_wn,
        )
        acc_gate += tl.dot(x_tile, w_gate_tile)
        acc_up += tl.dot(x_tile, w_up_tile)

    result = acc_gate * (1.0 / (1.0 + tl.exp(-acc_gate))) * acc_up

    tl.store(
        out_ptr + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on,
        result.to(tl.bfloat16),
        mask=row_mask[:, None] & n_mask[None, :],
    )


class Model(nn.Module):
    def __init__(self, T_total, H, I, E, K):
        super().__init__()
        self.T_total, self.H, self.I, self.E, self.K = T_total, H, I, E, K
        self.W_gate = nn.Parameter(torch.empty(E, H, I, dtype=torch.bfloat16))
        self.W_up = nn.Parameter(torch.empty(E, H, I, dtype=torch.bfloat16))
        for p in self.parameters():
            p.data.zero_()

    def forward(self, hidden_states, expert_offsets):
        T_perm, H = hidden_states.shape
        out = torch.empty(T_perm, self.I, dtype=torch.bfloat16, device=hidden_states.device)

        BM = 128
        BN = 256
        BK = 32

        max_tokens = T_perm // self.E + BM
        max_m_tiles = (max_tokens + BM - 1) // BM
        n_tiles = (self.I + BN - 1) // BN

        grid = (self.E, max_m_tiles, n_tiles)

        _grouped_gemm_swiglu_kernel[grid](
            hidden_states, self.W_gate, self.W_up, expert_offsets, out,
            self.I, H,
            hidden_states.stride(0), hidden_states.stride(1),
            self.W_gate.stride(0), self.W_gate.stride(1), self.W_gate.stride(2),
            out.stride(0), out.stride(1),
            BM=BM, BN=BN, BK=BK,
        )
        return out


T_total = 32768
H = 4096
I = 1536
E = 128
K = 8


def _build_routing(T_total, E, K, device="cpu"):
    T_perm = T_total * K
    base = T_perm // E
    rem = T_perm - base * E
    counts = torch.full((E,), base, dtype=torch.int32, device=device)
    counts[:rem] += 1
    offsets = torch.zeros(E + 1, dtype=torch.int32, device=device)
    offsets[1:] = torch.cumsum(counts, dim=0)
    return offsets


def get_inputs():
    T_perm = T_total * K
    hidden_states = torch.randn(T_perm, H, dtype=torch.bfloat16) * 0.1
    expert_offsets = _build_routing(T_total, E, K)
    return [hidden_states, expert_offsets]


def get_init_inputs():
    return [T_total, H, I, E, K]
